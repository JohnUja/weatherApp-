heres my javascript document.addEventListener("DOMContentLoaded", () => {
  class Card {
    constructor(data) {
      this.data = data;
      this.element = this.createElement();
    }

    createElement() {
      const card = document.createElement("div");
      card.classList.add("card");

      const cityNameElement = document.createElement("h2");
      cityNameElement.textContent = this.data.city;

      const iconElement = document.createElement("img");
      iconElement.src =
        "http://openweathermap.org/img/w/" + this.data.icon + ".png";
      iconElement.alt = "Weather Icon";

      const temperatureElement = document.createElement("p");
      temperatureElement.textContent = `Temperature: ${this.data.temp}°C`;

      const descriptionElement = document.createElement("p");
      descriptionElement.textContent = `Description: ${this.data.description}`;

      card.appendChild(cityNameElement);
      card.appendChild(iconElement);
      card.appendChild(descriptionElement);
      card.appendChild(temperatureElement);

      return card;
    }
  }

  class CardCollection {
    constructor() {
      this.cards = [];
      this.cardContainer = document.querySelector("#card-content");
      this.navigationContainer = document.querySelector("#card-navigation");
      this.addToCollectionButton = document.querySelector("#add-to-collection");

      this.addToCollectionButton.addEventListener("click", () => {
        if (this.cards.length > 0) {
          const activeCard = this.cards[0];
          activeCard.element.removeChild(this.addToCollectionButton);
        }
      });
    }

    async generateNewCard(cityName) {
      try {
        const response = await fetch("/", {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({ cityName }),
        });

        if (!response.ok) {
          throw new Error("Error fetching weather data");
        }

        const data = await response.json();
        const newCard = new Card(data);

        this.addCard(newCard);
      } catch (error) {
        console.error("Error fetching weather data:", error);
      }
    }

    addCard(card) {
      this.cards.unshift(card);
      this.renderCards();
    }

    renderCards() {
      this.cardContainer.innerHTML = "";
      this.navigationContainer.innerHTML = "";

      this.cards.forEach((card, index) => {
        this.cardContainer.appendChild(card.element);

        if (index === 0) {
          const prevButton = document.createElement("button");
          prevButton.textContent = "<";
          prevButton.classList.add("nav-button");
          prevButton.addEventListener("click", () => {
            this.prevCard();
          });

          this.navigationContainer.appendChild(prevButton);
        }
      });
    }

    prevCard() {
      this.cards.push(this.cards.shift());
      this.renderCards();
    }
  }

  const form = document.querySelector("#weatherForm");

  form.addEventListener("submit", (e) => {
    e.preventDefault();
    const cityNameInput = document.querySelector("#cityNameInput");
    const cityName = cityNameInput.value.trim();

    if (cityName) {
      cardCollection.generateNewCard(cityName);
      cityNameInput.value = "";
    }
  });

  const cardCollection = new CardCollection();
});
